async function createPaymentOrder(reportType, amount, subjectName) {
  try {

function fitPhone() {
  const wrap = document.querySelector('.phone-wrap');
  const PHONE_W = 406;  // 390px + 8px border ? 2
  const PHONE_H = 860;  // 844px + 8px border ? 2
  const PAD_TOP = 52;   // top info label height
  const PAD_SIDE = 80;  // room for left dots + right buttons

  const availW = window.innerWidth  - PAD_SIDE;
  const availH = window.innerHeight - PAD_TOP;

  const scale = Math.min(availW / PHONE_W, availH / PHONE_H, 1.0);

  // Apply via CSS zoom (preserves layout flow, supported in all modern browsers)
  wrap.style.zoom = scale.toFixed(4);
}
